# Core library that talks to Cozmo directly over WiFi (no cloud, no original app needed).
pycozmo==0.8.0

# pycozmo imports the stdlib "chunk" module, which was removed in Python 3.13
# (it's still needed to read Cozmo's built-in sound bank files). This package
# is the official PyPI backport that restores it. If you ever run this on
# Python 3.12 or older, this line is harmless but unnecessary.
standard-chunk==3.13.0

# Web backend that exposes driving/animation/TTS/camera endpoints to the
# browser-based frontend (Stage 3).
flask==3.1.3

# Offline text-to-speech for "make Cozmo talk". Windows uses pyttsx3 (its
# built-in SAPI voices); Linux uses piper-tts instead, since pyttsx3's Linux
# backend needs a system-installed espeak binary, which requires apt/sudo --
# not available on Endless OS's read-only base system. piper-tts is a plain
# pip package with everything it needs bundled in. See tts.py.
pyttsx3==2.99; sys_platform == "win32"
piper-tts==1.7.0; sys_platform != "win32"

# Face detection on Cozmo's camera feed. Pinned to the 4.x line
# deliberately -- OpenCV 5.0 removed CascadeClassifier (the classic face
# detector used in face_detection.py) entirely in favor of a newer
# DNN-based one that needs its own separately-downloaded model file. 4.x
# is simpler, needs no extra download beyond this package, and is a much
# more battle-tested codepath. "headless" build avoids GUI library
# dependencies (Qt etc.) we don't need and that complicate installs.
opencv-python-headless==4.10.0.84
