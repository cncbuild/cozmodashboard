# Core library that talks to Cozmo directly over WiFi (no cloud, no original app needed).
pycozmo==0.8.0

# pycozmo imports the stdlib "chunk" module, which was removed in Python 3.13
# (it's still needed to read Cozmo's built-in sound bank files). This package
# is the official PyPI backport that restores it. If you ever run this on
# Python 3.12 or older, this line is harmless but unnecessary.
standard-chunk==3.13.0

# Web backend that exposes driving/animation/TTS/camera endpoints to the
# browser-based frontend (Stage 3).
flask==3.1.3

# Offline text-to-speech for "make Cozmo talk" -- uses Windows' built-in
# SAPI voices, no internet or cloud account needed. On Linux this falls
# back to espeak (see Stage 4 notes about moving to Endless OS).
pyttsx3==2.99
